namespace WinForms { public partial class Form1 : Form { private TcpListener server;
    public Form1()
    {
        InitializeComponent();
    }

    private async void btnServerStart_Click(object sender, EventArgs e)
    {
        // 1. 9000번 포트로 서버 대기 시작
        server = new TcpListener(IPAddress.Any, 9000);
        server.Start();
        AddLog("서버가 시작되었습니다. (Port: 9000)");

        // 2. 클라이언트 접속 비동기 대기
        while (true)
        {
            TcpClient client = await server.AcceptTcpClientAsync();
            AddLog("새로운 클라이언트 접속!");
            _ = Task.Run(() => HandleClient(client)); // 접속한 클라이언트 처리
        }
    }
    private async Task HandleClient(TcpClient client)
    {
        using (NetworkStream stream = client.GetStream())
        {
            byte[] buffer = new byte[1024];
            int bytesRead;

            while ((bytesRead = await stream.ReadAsync(buffer, 0, buffer.Length)) > 0)
            {
                string message = Encoding.UTF8.GetString(buffer, 0, bytesRead);
                AddLog($"[수신]: {message}");
            }
        }
    }
    // 스레드 안전한 UI 업데이트 함수
    private void AddLog(string msg)
    {
        if (listBox1.InvokeRequired)
        {
            listBox1.Invoke(new Action(() => listBox1.Items.Add(msg)));
        }
        else
        {
            listBox1.Items.Add(msg);
        }
    }
}
